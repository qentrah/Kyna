# Android ARM64 shell: Qt supplies the Activity, Gradle packaging, and lifecycle.
if(NOT ANDROID)
  message(FATAL_ERROR "The Android shell requires a Qt Android toolchain")
endif()
if(DEFINED ANDROID_PLATFORM_LEVEL AND ANDROID_PLATFORM_LEVEL LESS 28)
  message(FATAL_ERROR "Kyna requires Android API 28 or newer; configure with -DANDROID_PLATFORM=android-28")
endif()
find_package(Qt6 6.7 REQUIRED COMPONENTS Quick)
qt_policy(SET QTP0002 NEW)
set(KYNA_ANDROID_APP_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/assets" CACHE PATH "Directory containing main.ky and application modules")
set(KYNA_ANDROID_APP_FILES "main.ky" CACHE STRING "Explicit semicolon-separated bundled application file list")
set(KYNA_ANDROID_NOTICES_QRC "" CACHE FILEPATH "Explicit resource list containing verified dependency notices")
qt_add_executable(kyna_android_shell MANUAL_FINALIZATION src/entry/android_main.cpp)
target_link_libraries(kyna_android_shell PRIVATE kyna_embedding Qt6::Quick)
target_compile_definitions(kyna_android_shell PRIVATE KYNA_ANDROID_GUI_LIBRARY="$<TARGET_FILE_NAME:kyna_gui>")
set(KYNA_ANDROID_RESOURCE_FILES)
foreach(KYNA_ANDROID_FILE IN LISTS KYNA_ANDROID_APP_FILES)
  if(NOT EXISTS "${KYNA_ANDROID_APP_SOURCE}/${KYNA_ANDROID_FILE}")
    message(FATAL_ERROR "Missing Android application module: ${KYNA_ANDROID_FILE}")
  endif()
  list(APPEND KYNA_ANDROID_RESOURCE_FILES "${KYNA_ANDROID_APP_SOURCE}/${KYNA_ANDROID_FILE}")
endforeach()
set_target_properties(kyna_android_shell PROPERTIES QT_QML_ROOT_PATH "${CMAKE_CURRENT_SOURCE_DIR}/qml")
qt_add_resources(kyna_android_shell kyna_deployment PREFIX /deployment FILES qml/deployment.qml)
qt_add_resources(kyna_android_shell kyna_application PREFIX /kyna BASE "${KYNA_ANDROID_APP_SOURCE}" FILES ${KYNA_ANDROID_RESOURCE_FILES})
if(KYNA_ANDROID_NOTICES_QRC)
  if(NOT EXISTS "${KYNA_ANDROID_NOTICES_QRC}")
    message(FATAL_ERROR "Missing Android dependency notice resource list")
  endif()
  set_target_properties(kyna_android_shell PROPERTIES AUTORCC ON)
  target_sources(kyna_android_shell PRIVATE "${KYNA_ANDROID_NOTICES_QRC}")
endif()
math(EXPR KYNA_ANDROID_VERSION_CODE "${PROJECT_VERSION_MAJOR} * 1000000 + ${PROJECT_VERSION_MINOR} * 1000 + ${PROJECT_VERSION_PATCH}")
set_target_properties(kyna_android_shell PROPERTIES QT_ANDROID_PACKAGE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/android" QT_ANDROID_EXTRA_LIBS "$<TARGET_FILE:kyna_gui>" QT_ANDROID_PACKAGE_NAME "org.kyna.application" QT_ANDROID_MIN_SDK_VERSION 28 QT_ANDROID_TARGET_SDK_VERSION 35 QT_ANDROID_VERSION_NAME "${PROJECT_VERSION}" QT_ANDROID_VERSION_CODE "${KYNA_ANDROID_VERSION_CODE}")
qt_finalize_executable(kyna_android_shell)
